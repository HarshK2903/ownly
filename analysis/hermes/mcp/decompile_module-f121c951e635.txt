export function getAuthErrorMessage(ok, arg1) {
  if (ok.ok) {
    return "";
  } else {
    const response = ok.response;
    let data;
    if (response != null) {
      data = response.data;
    }
    let message;
    if (data) {
      if (typeof data.message === "string") {
        message = data.message;
      }
    }
    if (message == null) {
      message = ok.error.message;
    }
    if (message == null) {
      message = arg1;
    }
    return message;
  }
}
